   123→function Workspace() {
   124→  const params = useParams<{ id: string }>();
   125→  const id = params?.id;
   126→  const router = useRouter();
   127→
   128→  const [project, setProject] = React.useState<ProjectDetail | null>(null);
   129→  const [loading, setLoading] = React.useState(true);
   130→  const [error, setError] = React.useState<string | null>(null);
   131→
   132→  const load = React.useCallback(async () => {
   133→    if (!id) return;
   134→    try {
   135→      const res = await fetch(`/api/projects/${id}`, { cache: "no-store" });
   136→      if (!res.ok) {
   137→        const err = await res.json().catch(() => ({}));
   138→        throw new Error(err.error || `HTTP ${res.status}`);
   139→      }
   140→      const data = await res.json();
   141→      setProject(data.project);
   142→      setError(null);
   143→    } catch (err) {
   144→      setError((err as Error).message);
   145→    } finally {
   146→      setLoading(false);
   147→    }
   148→  }, [id]);
   149→
   150→  React.useEffect(() => {
   151→    load();
   152→  }, [load]);
   153→
   154→  if (loading) {
   155→    return (
   156→      <div className="mx-auto w-full max-w-7xl flex-1 px-4 py-6 sm:px-6 lg:px-8">
   157→        <WorkspaceSkeleton />
   158→      </div>
   159→    );
   160→  }
   161→
   162→  if (error || !project) {
   163→    return (
   164→      <div className="mx-auto w-full max-w-3xl flex-1 px-4 py-12 sm:px-6 lg:px-8">
   165→        <Card className="border-rose-500/30 bg-rose-500/5">
   166→          <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   167→            <div className="flex h-12 w-12 items-center justify-center rounded-full bg-rose-500/10 text-rose-600">
   168→              <AlertCircle className="h-6 w-6" />
   169→            </div>
   170→            <h2 className="text-lg font-semibold">Could not load project</h2>
   171→            <p className="max-w-md text-sm text-muted-foreground">
   172→              {error || "Project not found, or you don't have access to it."}
   173→            </p>
   174→            <Button variant="outline" onClick={() => router.push("/projects")}>
   175→              <ArrowLeft className="h-4 w-4" />
   176→              Back to Projects
   177→            </Button>
   178→          </CardContent>
   179→        </Card>
   180→      </div>
   181→    );
   182→  }
   183→
   184→  return (
   185→    <div className="mx-auto w-full max-w-7xl flex-1 px-4 py-6 sm:px-6 lg:px-8">
   186→      <ProjectWorkspace project={project} reload={load} setProject={setProject} />
   187→    </div>
   188→  );
   189→}

   217→
   218→function ProjectWorkspace({
   219→  project,
   220→  reload,
   221→  setProject,
   222→}: {
   223→  project: ProjectDetail;
   224→  reload: () => Promise<void>;
   225→  setProject: React.Dispatch<React.SetStateAction<ProjectDetail | null>>;
   226→}) {
   227→  const router = useRouter();
   228→  const [settingsOpen, setSettingsOpen] = React.useState(false);
   229→  const [vscodePathOpen, setVscodePathOpen] = React.useState(false);
   230→  const [commitOpen, setCommitOpen] = React.useState(false);
   231→  const [githubOpen, setGithubOpen] = React.useState(false);
   232→  const [vercelOpen, setVercelOpen] = React.useState(false);
   233→  const [nenoOpen, setNenoOpen] = React.useState(false);
   234→  const [buildLogOpen, setBuildLogOpen] = React.useState(false);
   235→  const [deployOpen, setDeployOpen] = React.useState(false);
   236→  const [logPanelOpen, setLogPanelOpen] = React.useState(true);
   237→  const [busyAction, setBusyAction] = React.useState<string | null>(null);
   238→
   239→  // Local copy of the project name for inline editing.
   240→  const [editingName, setEditingName] = React.useState(false);
   241→  const [nameDraft, setNameDraft] = React.useState(project.name);
   242→
   243→  React.useEffect(() => setNameDraft(project.name), [project.name]);
   244→
   245→  const updateProject = (patch: Partial<ProjectDetail>) =>
   246→    setProject((prev) => (prev ? { ...prev, ...patch } : prev));
   247→
   248→  // ---------------------- VS Code deep-link ----------------------
   249→  const openInVSCode = () => {
   250→    if (!project.localPath) {
   251→      setVscodePathOpen(true);
   252→      return;
   253→    }
   254→    const url = `vscode://file/${encodeURIComponent(project.localPath).replace(/%2F/g, "/").replace(/%3A/g, ":")}`;
   255→    toast.info("Opening VS Code...", {
   256→      description: `Launching vscode:// with folder ${project.localPath}`,
   257→    });
   258→    // Use a hidden anchor to trigger the protocol handler.
   259→    const a = document.createElement("a");
   260→    a.href = url;
   261→    a.style.display = "none";
   262→    document.body.appendChild(a);
   263→    a.click();
   264→    document.body.removeChild(a);
   265→  };
   266→
   267→  // ---------------------- Build ----------------------
   268→  const triggerBuild = async () => {
   269→    setBusyAction("build");
   270→    try {
   271→      const res = await fetch(`/api/projects/${project.id}/build`, {
   272→        method: "POST",
   273→      });
   274→      if (!res.ok) {
   275→        const err = await res.json().catch(() => ({}));
   276→        throw new Error(err.error || "Build failed to start");
   277→      }
   278→      const data = await res.json();
   279→      // Optimistic update — show building state.
   280→      updateProject({
   281→        status: "ready", // post-build status from server
   282→        lastBuildAt: new Date().toISOString(),
   283→        builds: [data.buildRun, ...project.builds].slice(0, 10),
   284→      });
   285→      toast.success(
   286→        data.buildRun.status === "success"
   287→          ? "Build succeeded"
   288→          : "Build failed",
   289→        {
   290→          description:
   291→            data.buildRun.status === "success"
   292→              ? `Completed in ${(data.buildRun.duration / 1000).toFixed(2)}s`
   293→              : "Check the build log for details.",
   294→        }
   295→      );
   296→      setLogPanelOpen(true);
   297→      await reload();
   298→    } catch (err) {
   299→      toast.error("Build failed", { description: (err as Error).message });
   300→    } finally {
   301→      setBusyAction(null);
   302→    }
   303→  };
   304→
   305→  // ---------------------- Push to GitHub ----------------------
   306→  const pushToGitHub = async () => {
   307→    setBusyAction("push");
   308→    try {
   309→      // Simulate a push.
   310→      await new Promise((r) => setTimeout(r, 1500));
   311→      updateProject({ status: "github_synced" });
   312→      toast.success("Pushed to GitHub", {
   313→        description: project.githubRepo || "Changes pushed to default branch.",
   314→      });
   315→    } catch (err) {
   316→      toast.error("Push failed", { description: (err as Error).message });
   317→    } finally {
   318→      setBusyAction(null);
   319→    }
   320→  };
   321→
   322→  // ---------------------- Deploy ----------------------
   323→  const triggerDeploy = async () => {
   324→    setBusyAction("deploy");
   325→    try {
   326→      // Simulate deployment.
   327→      updateProject({ status: "deploying" });
   328→      await new Promise((r) => setTimeout(r, 2000));
   329→      const url = `https://${project.name.toLowerCase().replace(/[^a-z0-9-]/g, "-")}.vercel.app`;
   330→      // Persist lastDeployUrl.
   331→      await fetch(`/api/projects/${project.id}`, {
   332→        method: "PATCH",
   333→        headers: { "Content-Type": "application/json" },
   334→        body: JSON.stringify({ lastDeployUrl: url, status: "live" }),
   335→      });
   336→      updateProject({ status: "live", lastDeployUrl: url });
   337→      toast.success("Deployment live!", { description: url });
   338→    } catch (err) {
   339→      toast.error("Deploy failed", { description: (err as Error).message });
   340→      updateProject({ status: "error" });
   341→    } finally {
   342→      setBusyAction(null);
   343→    }
   344→  };
   345→
   346→  // ---------------------- Save name ----------------------
   347→  const saveName = async () => {
   348→    if (!nameDraft.trim() || nameDraft === project.name) {
   349→      setEditingName(false);
   350→      return;
   351→    }
   352→    setBusyAction("rename");
   353→    try {
   354→      const res = await fetch(`/api/projects/${project.id}`, {
   355→        method: "PATCH",
   356→        headers: { "Content-Type": "application/json" },
   357→        body: JSON.stringify({ name: nameDraft.trim() }),
   358→      });
   359→      if (!res.ok) throw new Error("Failed to rename");
   360→      updateProject({ name: nameDraft.trim() });
   361→      toast.success("Project renamed");
   362→    } catch (err) {
   363→      toast.error("Rename failed", { description: (err as Error).message });
   364→      setNameDraft(project.name);
   365→    } finally {
   366→      setBusyAction(null);
   367→      setEditingName(false);
   368→    }
   369→  };
   370→
   371→  // ---------------------- Delete ----------------------
   372→  const handleDelete = async () => {
   373→    setBusyAction("delete");
   374→    try {
   375→      const res = await fetch(`/api/projects/${project.id}`, {
   376→        method: "DELETE",
   377→      });
   378→      if (!res.ok) throw new Error("Failed to delete");
   379→      toast.success("Project deleted");
   380→      router.push("/projects");
   381→    } catch (err) {
   382→      toast.error("Delete failed", { description: (err as Error).message });
   383→      setBusyAction(null);
   384→    }
   385→  };
   386→
   387→  const latestBuild = project.builds[0] || null;
   388→  const meta = frameworkMeta(project.framework);
   389→
   390→  return (
   391→    <div className="flex flex-col gap-5">
   392→      {/* Back link */}
   393→      <Link
   394→        href="/projects"
   395→        className="inline-flex w-fit items-center gap-1 text-xs text-muted-foreground transition-colors hover:text-foreground"
   396→      >
   397→        <ArrowLeft className="h-3.5 w-3.5" />
   398→        All projects
   399→      </Link>
   400→
   401→      {/* ---------- Top bar ---------- */}
   402→      <Card className="overflow-hidden py-0">
   403→        <CardContent className="flex flex-col gap-4 p-5 lg:flex-row lg:items-center lg:justify-between">
   404→          <div className="flex min-w-0 items-center gap-3">
   405→            <div
   406→              className={cn(
   407→                "flex h-11 w-11 shrink-0 items-center justify-center rounded-xl text-base font-bold shadow-sm",
   408→                meta.color
   409→              )}
   410→              aria-hidden
   411→            >
   412→              {meta.icon}
   413→            </div>
   414→            <div className="min-w-0">
   415→              <div className="flex items-center gap-2">
   416→                {editingName ? (
   417→                  <div className="flex items-center gap-2">
   418→                    <Input
   419→                      value={nameDraft}
   420→                      onChange={(e) => setNameDraft(e.target.value)}
   421→                      className="h-7 w-48 text-base font-semibold"
   422→                      autoFocus
   423→                      onKeyDown={(e) => {
   424→                        if (e.key === "Enter") saveName();
   425→                        if (e.key === "Escape") {
   426→                          setNameDraft(project.name);
   427→                          setEditingName(false);
   428→                        }
   429→                      }}
   430→                    />
   431→                    <Button size="icon" variant="ghost" className="h-7 w-7" onClick={saveName}>
   432→                      <Check className="h-3.5 w-3.5" />
   433→                    </Button>
   434→                    <Button
   435→                      size="icon"
   436→                      variant="ghost"
   437→                      className="h-7 w-7"
   438→                      onClick={() => {
   439→                        setNameDraft(project.name);
   440→                        setEditingName(false);
   441→                      }}
   442→                    >
   443→                      <X className="h-3.5 w-3.5" />
   444→                    </Button>
   445→                  </div>
   446→                ) : (
   447→                  <>
   448→                    <h1 className="truncate text-xl font-semibold tracking-tight">
   449→                      {project.name}
   450→                    </h1>
   451→                    <TooltipProvider>
   452→                      <Tooltip>
   453→                        <TooltipTrigger asChild>
   454→                          <button
   455→                            onClick={() => setEditingName(true)}
   456→                            className="rounded-md p-1 text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   457→                            aria-label="Rename project"
   458→                          >
   459→                            <Pencil className="h-3.5 w-3.5" />
   460→                          </button>
   461→                        </TooltipTrigger>
   462→                        <TooltipContent>Rename</TooltipContent>
   463→                      </Tooltip>
   464→                    </TooltipProvider>
   465→                  </>
   466→                )}
   467→              </div>
   468→              <div className="mt-1 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   469→                <Badge variant="outline" className="h-5 text-[10px]">
   470→                  {meta.label}
   471→                </Badge>
   472→                <Badge
   473→                  variant="outline"
   474→                  className={cn("h-5 text-[10px]", statusBadgeClass(project.status))}
   475→                >
   476→                  {project.status === "live" && (
   477→                    <span className="mr-1 inline-block h-1.5 w-1.5 animate-pulse rounded-full bg-emerald-500" />
   478→                  )}
   479→                  {statusLabel(project.status)}
   480→                </Badge>
   481→                {project.localPath && (
   482→                  <span className="inline-flex items-center gap-1 font-mono">
   483→                    <Folder className="h-3 w-3" />
   484→                    {project.localPath}
   485→                  </span>
   486→                )}
   487→                <span>· Updated {formatRelative(project.updatedAt)}</span>
   488→              </div>
   489→            </div>
   490→          </div>
   491→
   492→          <div className="flex flex-wrap items-center gap-2">
   493→            <TooltipProvider>
   494→              <Tooltip>
   495→                <TooltipTrigger asChild>
   496→                  <Button onClick={openInVSCode} variant="default">
   497→                    <Code className="h-4 w-4" />
   498→                    Open in VS Code
   499→                  </Button>
   500→                </TooltipTrigger>
   501→                <TooltipContent>
   502→                  {project.localPath
   503→                    ? `vscode://file/${project.localPath}`
   504→                    : "Set your local path first"}
   505→                </TooltipContent>
   506→              </Tooltip>
   507→            </TooltipProvider>
   508→            <Button variant="outline" onClick={() => setSettingsOpen(true)}>
   509→              <SettingsIcon className="h-4 w-4" />
   510→              Settings
   511→            </Button>
   512→          </div>
   513→        </CardContent>
   514→      </Card>
   515→
   516→      {/* ---------- Status flow stepper ---------- */}
   517→      <StatusFlowStepper status={project.status} />
   518→
   519→      {/* ---------- Action grid ---------- */}
   520→      <div>
   521→        <h2 className="mb-3 text-sm font-semibold text-muted-foreground">
   522→          Lifecycle actions
   523→        </h2>
   524→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   525→          <ActionButton
   526→            icon={FolderOpen}
   527→            label="Open Local Folder"
   528→            sublabel={project.localPath ? "Path set" : "Set local path"}
   529→            onClick={() => setVscodePathOpen(true)}
   530→          />
   531→          <ActionButton
   532→            icon={Code}
   533→            label="Open in VS Code"
   534→            sublabel={project.localPath ? "Deep-link ready" : "Set local path"}
   535→            disabled={!project.localPath}
   536→            onClick={openInVSCode}
   537→            accent="blue"
   538→          />
   539→          <ActionButton
   540→            icon={Github}
   541→            label="Connect GitHub"
   542→            sublabel={project.githubConnected ? "Connected" : "Not connected"}
   543→            connected={project.githubConnected}
   544→            onClick={() => setGithubOpen(true)}
   545→          />
   546→          <ActionButton
   547→            icon={GitCommit}
   548→            label="Commit Changes"
   549→            sublabel="Write a commit"
   550→            onClick={() => setCommitOpen(true)}
   551→          />
   552→          <ActionButton
   553→            icon={GitBranch}
   554→            label="Push to GitHub"
   555→            sublabel={project.githubConnected ? "Push to remote" : "GitHub required"}
   556→            disabled={!project.githubConnected}
   557→            loading={busyAction === "push"}
   558→            onClick={pushToGitHub}
   559→          />
   560→          <ActionButton
   561→            icon={Zap}
   562→            label="Connect Neno"
   563→            sublabel={project.nenoConnected ? "Connected" : "Not connected"}
   564→            connected={project.nenoConnected}
   565→            onClick={() => setNenoOpen(true)}
   566→          />
   567→          <ActionButton
   568→            icon={Hammer}
   569→            label="Build Project"
   570→            sublabel={latestBuild ? `Last: ${statusLabel(latestBuild.status)}` : "Never built"}
   571→            loading={busyAction === "build"}
   572→            onClick={triggerBuild}
   573→            accent="amber"
   574→          />
   575→          <ActionButton
   576→            icon={Rocket}
   577→            label="Connect Vercel"
   578→            sublabel={project.vercelConnected ? "Connected" : "Not connected"}
   579→            connected={project.vercelConnected}
   580→            onClick={() => setVercelOpen(true)}
   581→          />
   582→          <ActionButton
   583→            icon={Cloud}
   584→            label="Deploy to Vercel"
   585→            sublabel={
   586→              project.vercelConnected
   587→                ? project.lastDeployUrl
   588→                  ? "Re-deploy"
   589→                  : "Deploy now"
   590→                : "Vercel required"
   591→            }
   592→            disabled={!project.vercelConnected}
   593→            loading={busyAction === "deploy"}
   594→            onClick={triggerDeploy}
   595→            accent="indigo"
   596→          />
   597→          <ActionButton
   598→            icon={Terminal}
   599→            label="View Build Logs"
   600→            sublabel={latestBuild ? formatRelative(latestBuild.createdAt) : "No builds yet"}
   601→            disabled={!latestBuild}
   602→            onClick={() => setBuildLogOpen(true)}
   603→          />
   604→          <ActionButton
   605→            icon={Globe}
   606→            label="Deployment Status"
   607→            sublabel={project.lastDeployUrl ? "Live" : "Not deployed"}
   608→            onClick={() => setDeployOpen(true)}
   609→          />

   727→const STAGES = [
   728→  { key: "project", label: "Project", icon: Folder, status: "development" },
   729→  { key: "development", label: "Development", icon: Code, status: "development" },
   730→  { key: "changes", label: "Changes Detected", icon: GitBranch, status: "changes_detected" },
   731→  { key: "github", label: "GitHub Sync", icon: Github, status: "github_synced" },
   732→  { key: "build", label: "Build", icon: Hammer, status: "building" },
   733→  { key: "ready", label: "Ready", icon: CheckCircle2, status: "ready" },
   734→  { key: "deploying", label: "Deployment", icon: Rocket, status: "deploying" },
   735→  { key: "live", label: "Live", icon: Globe, status: "live" },
   736→];
   737→
   738→const STATUS_ORDER: Record<string, number> = {
   739→  development: 1,
   740→  changes_detected: 2,
   741→  github_synced: 3,
   742→  building: 4,
   743→  ready: 5,
   744→  deploying: 6,
   745→  live: 7,
   746→  error: 0,
   747→};
   748→
   749→function StatusFlowStepper({ status }: { status: string }) {
   750→  const currentIdx = STATUS_ORDER[status] ?? 0;
   751→  return (
   752→    <Card className="py-0">
   753→      <CardContent className="overflow-x-auto p-4">
   754→        <div className="flex min-w-max items-center gap-1">
   755→          {STAGES.map((stage, i) => {
   756→            const stageIdx = i + 1; // project = 1
   757→            const isDone = stageIdx < currentIdx;
   758→            const isActive = stageIdx === currentIdx;
   759→            const isPending = stageIdx > currentIdx;
   760→            const Icon = stage.icon;
   761→            return (
   762→              <React.Fragment key={stage.key}>
   763→                <div className="flex flex-col items-center gap-1.5">
   764→                  <div
   765→                    className={cn(
   766→                      "flex h-10 w-10 items-center justify-center rounded-full border-2 transition-all",
   767→                      isDone && "border-emerald-500 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400",
   768→                      isActive && "border-blue-500 bg-blue-500/10 text-blue-600 dark:text-blue-400 animate-pulse",
   769→                      isPending && "border-border bg-muted text-muted-foreground"
   770→                    )}
   771→                  >
   772→                    <Icon className="h-4 w-4" />
   773→                  </div>
   774→                  <span
   775→                    className={cn(
   776→                      "text-[10px] font-medium whitespace-nowrap",
   777→                      isActive && "text-blue-600 dark:text-blue-400",
   778→                      isDone && "text-emerald-600 dark:text-emerald-400",
   779→                      isPending && "text-muted-foreground"
   780→                    )}
   781→                  >
   782→                    {stage.label}
   783→                  </span>
   784→                </div>
   785→                {i < STAGES.length - 1 && (
   786→                  <div
   787→                    className={cn(
   788→                      "mb-5 h-0.5 w-8 shrink-0 rounded-full transition-colors sm:w-12",
   789→                      i + 1 < currentIdx
   790→                        ? "bg-emerald-500/60"
   791→                        : "bg-border"
   792→                    )}
   793→                  />
   794→                )}
   795→              </React.Fragment>
   796→            );
   797→          })}
   798→        </div>
   799→      </CardContent>
   800→    </Card>
   801→  );
   802→}

   989→function FileTreeNode({
   990→  node,
   991→  depth,
   992→  selected,
   993→  onSelect,
   994→}: {
   995→  node: FileNode;
   996→  depth: number;
   997→  selected: FileNode | null;
   998→  onSelect: (n: FileNode) => void;
   999→}) {
  1000→  const [open, setOpen] = React.useState(depth < 2);
  1001→  if (node.name === "root") {
  1002→    return (
  1003→      <div>
  1004→        {node.children?.map((c) => (
  1005→          <FileTreeNode
  1006→            key={c.name}
  1007→            node={c}
  1008→            depth={depth}
  1009→            selected={selected}
  1010→            onSelect={onSelect}
  1011→          />
  1012→        ))}
  1013→      </div>
  1014→    );
  1015→  }
  1016→  if (node.type === "dir") {
  1017→    return (
  1018→      <div>
  1019→        <button

  1021→          className="flex w-full items-center gap-1 rounded-md px-1.5 py-1 text-left text-xs hover:bg-accent"

  1030→            <Folder className="h-3.5 w-3.5 text-amber-500" />
  1031→          )}

  1033→        </button>

  1035→          <div>
  1036→            {node.children?.map((c) => (
  1037→              <FileTreeNode
  1038→                key={c.name}
  1039→                node={c}
  1040→                depth={depth + 1}
  1041→                selected={selected}
  1042→                onSelect={onSelect}
  1043→              />
  1044→            ))}
  1045→          </div>
  1046→        )}
  1047→      </div>

  1052→      onClick={() => onSelect(node)}
